ESQUEMA DE LA BASE DE DATOS - Sistema de gestión de la veterinaria
Motor: MySQL 8  |  Base: veterinaria

Este archivo es solo DOCUMENTACIÓN del esquema.
La fuente real es InicializadorBD.java, que crea todo automáticamente

---
CREATE DATABASE IF NOT EXISTS veterinaria CHARACTER SET utf8mb4;
USE veterinaria;


cliente: dueños de las mascotas
---
CREATE TABLE IF NOT EXISTS cliente (
    id        INT AUTO_INCREMENT PRIMARY KEY,
    nombre    VARCHAR(60) NOT NULL,
    dni       VARCHAR(20) NOT NULL UNIQUE,
    telefono  VARCHAR(30) NOT NULL
);

mascota: pertenece a un cliente (relación 1 cliente : N mascotas)
El nombre puede ser NULL (porque pueden llevar animales de la calle o muy chicos que no tengan nombre)
---
CREATE TABLE IF NOT EXISTS mascota (
    id          INT AUTO_INCREMENT PRIMARY KEY,
    nombre      VARCHAR(60),
    especie     VARCHAR(50) NOT NULL,
    cliente_id  INT NOT NULL,
    FOREIGN KEY (cliente_id) REFERENCES cliente(id)
);


profesional: veterinarios
especialidad guarda el nombre del enum Especialidad: GENERAL | CIRUGIA | VACUNACION
---
CREATE TABLE IF NOT EXISTS profesional (
    id            INT AUTO_INCREMENT PRIMARY KEY,
    matricula     INT NOT NULL UNIQUE,
    nombre        VARCHAR(40) NOT NULL,
    apellido      VARCHAR(40) NOT NULL,
    telefono      VARCHAR(30),
    correo        VARCHAR(60),
    especialidad  VARCHAR(20) NOT NULL
);

servicio: prestaciones que ofrece la veterinaria
duracion_min: duracion estimada fija en minutos
especialidad_requerida: especialidad que debe tener el profesional
---
CREATE TABLE IF NOT EXISTS servicio (
    id                      INT AUTO_INCREMENT PRIMARY KEY,
    nombre                  VARCHAR(60) NOT NULL UNIQUE,
    duracion_min            INT NOT NULL,
    especialidad_requerida  VARCHAR(20) NOT NULL
);

turno: une una mascota, un profesional y un servicio
duracion_est: duración estimada en minutos
estado hace referencia al enum EstadoTurno: PROGRAMADO | CANCELADO | FINALIZADO
---
CREATE TABLE IF NOT EXISTS turno (
    id              INT AUTO_INCREMENT PRIMARY KEY,
    mascota_id      INT NOT NULL,
    profesional_id  INT NOT NULL,
    servicio_id     INT NOT NULL,
    fecha           DATE NOT NULL,
    horario         TIME NOT NULL,
    duracion_est    INT NOT NULL,
    estado          VARCHAR(15) NOT NULL DEFAULT 'PROGRAMADO',
    FOREIGN KEY (mascota_id)     REFERENCES mascota(id),
    FOREIGN KEY (profesional_id) REFERENCES profesional(id),
    FOREIGN KEY (servicio_id)    REFERENCES servicio(id)
);
